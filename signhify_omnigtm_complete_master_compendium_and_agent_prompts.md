class ConfigurationError(RuntimeError):

SECRET_REDACTION_REGEX = re.compile(
    r"(postgres(?:ql)?://[^\s]+|sk-[a-zA-Z0-9_-]{12,}|AIza[0-9A-Za-z-_]{15,})"

def scrub_secrets(text: str) -> str:
  """Redacts database URIs and API keys from any text string."""
  return SECRET_REDACTION_REGEX.sub("[REDACTED_SECRET]", text)


def require_secret(env_key: str, allow_offline_mock: bool = False) -> str:
  """Strictly fetches an environment variable without hardcoded secret fallbacks.

  Matches the Phase 0 security fix in Warriorlegacy/Autogram#1.
  """

    raise ConfigurationError(
        f"Missing required environment variable '{env_key}'. "
        "Plaintext secret fallbacks are strictly prohibited."
    )
  return value

class ZeroCostMultiModelRouter:
  """Routes structured prompts across verified ₹0 LLM tiers with circuit-breaking."""

  def __init__(self, offline_mode: bool = False) -> None:
    self.offline_mode = (
        offline_mode or os.environ.get("OMNIGTM_OFFLINE_TEST") == "1"
    )
    self.providers = ZERO_BUDGET_PROVIDER_CHAIN
    self._simulated_failures: set[str] = set()

  def simulate_provider_rate_limit(self, provider_name: str) -> None:
    self._simulated_failures.add(provider_name)

  def generate_structured(
      self,
      task_type: str,
      prompt: str,
      context_chunks: List[str],
      custom_adapter: Optional[Callable[[ProviderSpec, str], Dict[str, Any]]] = None,
  ) -> GenerationResult:
    start_ts = time.perf_counter()
    attempts = 0
    last_error: Optional[Exception] = None

    for spec in self.providers:
      attempts += 1
      try:
        _ = require_secret(spec.env_key, allow_offline_mock=self.offline_mode)
        if spec.name in self._simulated_failures:
          raise RuntimeError(f"HTTP 429 Rate Limit Exceeded on {spec.name}")

        if custom_adapter is not None:
          payload = custom_adapter(spec, prompt)
        else:

              spec=spec,
